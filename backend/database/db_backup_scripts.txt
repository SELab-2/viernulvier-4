CREATE TABLE productions (
    titel TEXT,
    ondertitel TEXT,
    description1 TEXT,
    description2 TEXT,
    genre TEXT,
    id SERIAL PRIMARY KEY,
    planning_id INT,
    blog_titel TEXT,
    blog_text TEXT
);

CREATE TABLE events (
    id SERIAL PRIMARY KEY,
    starttime DATE NOT NULL,
    endtime DATE NOT NULL,
    hall TEXT,
    production_id INT NOT NULL,
    CONSTRAINT fk_production
        FOREIGN KEY (production_id)
            REFERENCES productions(id)
            ON DELETE CASCADE
);
